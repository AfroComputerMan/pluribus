# Responsive Sidebar Drawer Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Replace the `display: none` responsive behaviour at `< 768px` with a slide-over Sheet drawer that contains the ThreadList and a merged Notes section, triggered by a hamburger button in the chat header.

**Architecture:** Install the shadcn `Sheet` component. Add a `useMediaQuery` hook to `ThreadWorkspace.tsx` that detects `< 768px`. When narrow, render a `Sheet` from the left containing `<ThreadList />` (scrollable) and the Notes textarea below a separator; the existing 3-column grid layout is used unchanged above 768px.

**Tech Stack:** shadcn/ui Sheet (Radix Dialog under the hood), React `useState` + `useEffect` for the media query hook, Tailwind CSS v4.

### Task 2: Add `useMediaQuery` hook

**Files:**
- Create: `src/hooks/use-media-query.ts`

**Step 1: Write the hook**

```ts
// src/hooks/use-media-query.ts
import { useEffect, useState } from "react";

export function useMediaQuery(query: string): boolean {
  const [matches, setMatches] = useState(() => window.matchMedia(query).matches);

  useEffect(() => {
    const media = window.matchMedia(query);
    const listener = (event: MediaQueryListEvent) => setMatches(event.matches);
    media.addEventListener("change", listener);
    setMatches(media.matches);
    return () => media.removeEventListener("change", listener);
  }, [query]);

  return matches;
}
```

### Task 3: Rewrite `ThreadWorkspace.tsx` for responsive drawer layout

**Files:**
- Modify: `src/components/ThreadWorkspace.tsx`

**Step 1: Replace the file content**

Replace the entire contents of `src/components/ThreadWorkspace.tsx` with:

```tsx
import { Thread } from "@/components/assistant-ui/thread";
import { ThreadList } from "@/components/assistant-ui/thread-list";
import { Button } from "@/components/ui/button";
import {
  Sheet,
  SheetContent,
  SheetHeader,
  SheetTitle,
} from "@/components/ui/sheet";
import { useMediaQuery } from "@/hooks/use-media-query";
import { cn } from "@/lib/utils";
import {
  MenuIcon,
  PanelLeftCloseIcon,
  PanelLeftOpenIcon,
  PanelRightCloseIcon,
  PanelRightOpenIcon,
} from "lucide-react";
import { useState } from "react";

export function ThreadWorkspace() {
  const [isNavOpen, setIsNavOpen] = useState(true);
  const [isUtilityOpen, setIsUtilityOpen] = useState(true);
  const [isDrawerOpen, setIsDrawerOpen] = useState(false);
  const isNarrow = useMediaQuery("(max-width: 767px)");

  const shellClassName = "app-shell h-full w-full";

  if (isNarrow) {
    return (
      <div className={shellClassName}>
        {/* Narrow layout: full-width chat + hamburger drawer trigger */}
        <div className="workspace-grid workspace-narrow">
          <section className="workspace-thread">
            <Button
              variant="ghost"
              size="icon-xs"
              className="workspace-panel-toggle absolute top-2 left-2 z-20"
              onClick={() => setIsDrawerOpen(true)}
              aria-label="Open sidebar"
            >
              <MenuIcon className="size-4" />
            </Button>
            <Thread />
          </section>
        </div>

        <Sheet open={isDrawerOpen} onOpenChange={setIsDrawerOpen}>
          <SheetContent
            side="left"
            className="workspace-drawer-sheet flex flex-col p-0 w-[280px]"
          >
            <SheetHeader className="sr-only">
              <SheetTitle>Navigation</SheetTitle>
            </SheetHeader>
            {/* Thread list — scrolls independently */}
            <div className="min-h-0 flex-1 overflow-hidden">
              <ThreadList />
            </div>
            {/* Notes — merged into drawer */}
            <div className="workspace-drawer-notes border-t border-border">
              <p className="workspace-utility-label px-3 pt-3 pb-1">Notes</p>
              <textarea
                className="workspace-utility-notes mx-3 mb-3"
                style={{ minHeight: "8rem", width: "calc(100% - 1.5rem)" }}
                placeholder="Capture scratch notes, commands, and reminders..."
              />
            </div>
          </SheetContent>
        </Sheet>
      </div>
    );
  }

  return (
    <div className={shellClassName}>
      <div
        className={cn(
          "workspace-grid",
          !isNavOpen && "nav-collapsed",
          !isUtilityOpen && "utility-collapsed",
        )}
      >
        <aside className={cn("workspace-nav", !isNavOpen && "collapsed")}>
          <div className="workspace-nav-header">
            <Button
              variant="ghost"
              size="icon-xs"
              className="workspace-panel-toggle"
              onClick={() => setIsNavOpen(false)}
              aria-label="Collapse sidebar"
            >
              <PanelLeftCloseIcon className="size-4" />
            </Button>
          </div>
          <div className="workspace-nav-content">
            <ThreadList />
          </div>
        </aside>
        <section className="workspace-thread">
          {!isNavOpen ? (
            <Button
              variant="ghost"
              size="icon-xs"
              className="workspace-panel-toggle absolute top-2 left-2 z-20"
              onClick={() => setIsNavOpen(true)}
              aria-label="Expand sidebar"
            >
              <PanelLeftOpenIcon className="size-4" />
            </Button>
          ) : null}
          {!isUtilityOpen ? (
            <Button
              variant="ghost"
              size="icon-xs"
              className="workspace-panel-toggle absolute top-2 right-2 z-20"
              onClick={() => setIsUtilityOpen(true)}
              aria-label="Expand notes panel"
            >
              <PanelRightOpenIcon className="size-4" />
            </Button>
          ) : null}
          <Thread />
        </section>
        <aside className={cn("workspace-utility", !isUtilityOpen && "collapsed")}>
          <div className="workspace-utility-header">
            <Button
              variant="ghost"
              size="icon-xs"
              className="workspace-panel-toggle"
              onClick={() => setIsUtilityOpen(false)}
              aria-label="Collapse notes panel"
            >
              <PanelRightCloseIcon className="size-4" />
            </Button>
          </div>
          <div className="workspace-utility-content">
            <div className="workspace-utility-body">
              <p className="workspace-utility-label">Notes</p>
              <textarea
                className="workspace-utility-notes"
                placeholder="Capture scratch notes, commands, and reminders..."
              />
            </div>
          </div>
        </aside>
      </div>
    </div>
  );
}
```
